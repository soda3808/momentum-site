// ============================================================================
// Kensho Free Session Background (OPEN SOURCE · community)
// Purpose : Light background for Asia / London / New York sessions.
//           Teaching aid only — not signals, not a strategy, not edge.
// Best on : M5 – H1
// ============================================================================
//@version=6

//             Not a prop-firm / FTMO pass tool. Past patterns ≠ future results.

indicator("Kensho Free Session Background", overlay = true, max_boxes_count = 500)

// ── Defaults use exchange/chart timezone via input.session ───────────────────
grp_a = "Asia"
showA = input.bool(true, "Show Asia", group = grp_a)
sessA = input.session("2000-0400", "Asia session", group = grp_a)
colA  = input.color(color.new(#42a5f5, 88), "Asia color", group = grp_a)

grp_l = "London"
showL = input.bool(true, "Show London", group = grp_l)
sessL = input.session("0300-1130", "London session", group = grp_l)
colL  = input.color(color.new(#ab47bc, 88), "London color", group = grp_l)

grp_n = "New York"
showN = input.bool(true, "Show New York", group = grp_n)
sessN = input.session("0830-1600", "New York session", group = grp_n)
colN  = input.color(color.new(#ef5350, 88), "NY color", group = grp_n)

grp_x = "Display"
transp = input.int(88, "Background transparency", minval = 70, maxval = 95, group = grp_x)

inSess(s) =>
    not na(time(timeframe.period, s))

// Soft stack: if sessions overlap, last bgcolor wins — fine for teaching
bgcolor(showA and inSess(sessA) ? color.new(colA, transp) : na, title = "Asia")
bgcolor(showL and inSess(sessL) ? color.new(colL, transp) : na, title = "London")
bgcolor(showN and inSess(sessN) ? color.new(colN, transp) : na, title = "New York")

// Tiny label once per day so people know what they installed
var label tip = na
if barstate.islast and timeframe.isintraday
    label.delete(tip)
    tip := label.new(bar_index, high, "Sessions = context only", style = label.style_label_down, textcolor = color.white, color = color.new(color.gray, 30), size = size.tiny)
